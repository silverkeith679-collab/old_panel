install_require() {
  clear
  echo "Installing dependencies (Debian 11/12 compatible)..."
  {
    export DEBIAN_FRONTEND=noninteractive
    export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

    # --- Disable conflicting firewalls ---
    systemctl stop firewalld 2>/dev/null
    systemctl disable firewalld 2>/dev/null
    systemctl stop nftables 2>/dev/null
    systemctl disable nftables 2>/dev/null
    systemctl stop ufw 2>/dev/null
    systemctl disable ufw 2>/dev/null

    # --- Update repositories ---
    apt update -y
    apt install -y software-properties-common apt-transport-https

    # --- Core tools ---
    apt install -y curl wget git tar gzip unzip jq nano neofetch screen cron sudo lsof socat vnstat fail2ban dbus systemd iptables net-tools dnsutils mlocate build-essential

    # --- Networking & server tools ---
    apt install -y squid dropbear stunnel4 ca-certificates libssl-dev libffi-dev gnutls-bin netcat-openbsd httpie

    # --- PHP & Python ---
    apt install -y php php-cli php-curl php-mysql default-mysql-client python3 python3-pip python3-venv dos2unix

    # --- Virtualization detection ---
    apt install -y virt-what
    virt-what > /root/virt_type.txt 2>/dev/null

    # --- Cron setup ---
    systemctl enable cron >/dev/null 2>&1
    systemctl start cron >/dev/null 2>&1
    touch /var/spool/cron/crontabs/root
    chmod 600 /var/spool/cron/crontabs/root

    # --- Ensure directories exist ---
    mkdir -p /usr/local/bin
    mkdir -p /usr/local/var/run
    mkdir -p /root/.web

echo "HOST='$HOST'" >> /etc/.db-base
echo "DB='$DBNAME'" >> /etc/.db-base
echo "USER='$USER'" >> /etc/.db-base
echo "PASS='$PASS'" >> /etc/.db-base

  # --- Cleanup ---
    apt autoremove -y
    apt clean

    clear
    echo "✅ All dependencies installed successfully."
  } &>/var/log/install_require.log
}

install_firewall_kvm() {
    echo "→ Configuring firewall (iptables)..."
    sysctl -w net.ipv4.ip_forward=1
    sysctl -w net.ipv4.conf.all.rp_filter=0

    iptables -F
    iptables -A INPUT -p udp --dport 20100:20900 -m state --state NEW -m recent --set
    iptables -A INPUT -p udp --dport 5300 -j ACCEPT
    iptables -A INPUT -p tcp --dport 8080 -j ACCEPT

    mkdir -p /etc/iptables
    iptables-save > /etc/iptables/rules.v4
    ip6tables-save > /etc/iptables/rules.v6

    echo '* soft nofile 65536' >> /etc/security/limits.conf
    echo '* hard nofile 65536' >> /etc/security/limits.conf

    if systemctl list-units | grep -q netfilter-persistent; then
        systemctl enable netfilter-persistent
        systemctl restart netfilter-persistent
    else
        systemctl enable iptables-persistent || true
        systemctl restart iptables-persistent || true
    fi
}
